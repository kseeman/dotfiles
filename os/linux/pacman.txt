# Arch package manifest, mirroring os/macos/Brewfile — keep the two in sync
# when adding a tool. One package per line; blank lines and # comments ignored.
#
# Installed with: pacman -S --needed
# AUR packages belong in aur.txt instead. The Hyprland desktop's packages are
# its own repository's manifest, not this one.

git
neovim
# Required by nvim-treesitter's main branch to generate parsers. Must come from
# a package manager, NOT npm (per upstream requirements).
tree-sitter-cli
fzf
zoxide
fastfetch
bat
ripgrep
jq
tree
# Opened in a popup by tmux's prefix + g. Optional — that binding reports its
# absence rather than failing — but listed so a fresh machine has it.
lazygit
tmux
# Builds the venv the nvim python profile runs its Jupyter host from (see
# install.sh).
python
# image.nvim's `magick_cli` processor shells out to `magick` to resize images:
# mermaid diagrams, and Jupyter plot output in the python profile.
imagemagick

# nvm ships as scripts under /usr/share/nvm rather than a binary; the path is
# wired up in os/linux/zsh/exports.zsh. Node versions still install to ~/.nvm.
nvm

# Pulls in the system nodejs, which shadows nvm-managed node on PATH only if
# nvm isn't loaded. Same trade-off as `brew "pnpm"` on macOS.
pnpm

# Terminal. kitty/kitty.conf is linked by install.sh and expects the
# CaskaydiaCove Nerd Font family, which ttf-cascadia-code-nerd provides.
kitty
ttf-cascadia-code-nerd
